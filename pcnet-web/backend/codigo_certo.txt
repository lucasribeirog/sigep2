app.post('/api/gerar-laudo', upload.single('documento'), async (req, res) => {
    if (!req.file) {
        return res.status(400).json({ erro: 'Nenhum documento do PCNet foi enviado.' });
    }

    try {
        console.log('📂 Recebendo documento e processando variáveis...');
        
        // Caminho do seu modelo salvo que contém todos os blocos {#is_faca}, {#is_outro}, etc.
        const pathModelo = './modelo_faca_limpo.docx'; 
        if (!fs.existsSync(pathModelo)) {
            return res.status(500).json({ erro: 'Modelo mestre não encontrado no servidor.' });
        }
        
        let arquivoModeloBuffer = fs.readFileSync(pathModelo);
        const arquivoPCNetBuffer = req.file.buffer;

        // =========================================================
        // PASSO 1: PREENCHENDO O MODELO COM DOCXTEMPLATER
        // =========================================================
        const zipParaTemplater = new PizZip(arquivoModeloBuffer);
        const doc = new Docxtemplater(zipParaTemplater, {
            paragraphLoop: true,
            linebreaks: true,
        });

        // O tipo de objeto dita qual bloco {#is_...} será renderizado no Word
        const tipoObjeto = req.body.tipo_objeto || 'faca'; 

        doc.render({
            // Chaves Lógicas (Interruptores)
            is_faca: tipoObjeto === 'faca',
            is_canivete: tipoObjeto === 'canivete',
            is_madeira: tipoObjeto === 'madeira',
            is_outro: tipoObjeto === 'outro',

            // Variáveis Comuns (Faca / Canivete)
            marca: req.body.marca || "sem marca aparente",
            material_cabo: req.body.material_cabo || "não informado",
            cor_cabo: req.body.cor_cabo || "não informada",
            comp_lamina: req.body.comp_lamina || "XX",
            largura_base: req.body.largura_base || "XX",
            comp_total: req.body.comp_total || "XX",

            // Variáveis Específicas: Canivete
            tipo_abertura: req.body.tipo_abertura || "N/A",

            // Variáveis Específicas: Madeira
            secao_madeira: req.body.secao_madeira || "N/A",
            comp_madeira: req.body.comp_madeira || "XX",
            larg_madeira: req.body.larg_madeira || "XX",
            massa: req.body.massa || "XX",

            // Variáveis: O Bloco "Coringa" (Outros)
            nome_objeto: req.body.nome_objeto || "objeto não especificado",
            material_predominante: req.body.material_predominante || "não informado",
            cor_objeto: req.body.cor_objeto || "não informada",
            compr_objeto: req.body.compr_objeto || "XX",
            larg_objeto: req.body.larg_objeto || "XX",
            espessura_objeto: req.body.espessura_objeto || "XX",
            massa_objeto: req.body.massa_objeto || "XX"
        });

        // Gera o buffer do modelo preenchido sem as variáveis não utilizadas
        arquivoModeloBuffer = doc.getZip().generate({ type: 'nodebuffer' });
        console.log(`✍️ Modelo preenchido com sucesso para o tipo: ${tipoObjeto}`);

        // =========================================================
        // PASSO 2: A CIRURGIA XML (Corte e Transplante)
        // =========================================================
        const zipPCNet = new PizZip(arquivoPCNetBuffer);
        const zipModelo = new PizZip(arquivoModeloBuffer); 

        const xmlPCNetString = zipPCNet.file("word/document.xml").asText();
        const xmlModeloString = zipModelo.file("word/document.xml").asText();

        const parser = new DOMParser();
        const docPCNet = parser.parseFromString(xmlPCNetString, 'text/xml');
        const docModelo = parser.parseFromString(xmlModeloString, 'text/xml');

        const paragrafosPCNet = docPCNet.getElementsByTagName('w:p');
        let indexCorte = -1;

        for (let i = 0; i < paragrafosPCNet.length; i++) {
            if ((paragrafosPCNet[i].textContent || "").includes("Hora do início do exame")) {
                indexCorte = i;
                break;
            }
        }

        if (indexCorte === -1) {
            return res.status(400).json({ erro: 'Linha "Hora do início do exame" não encontrada no documento.' });
        }

        console.log('✂️ Ponto de corte encontrado! Removendo texto inútil do PCNet...');
        const bodyPCNet = docPCNet.getElementsByTagName('w:body')[0];
        
        for (let i = paragrafosPCNet.length - 1; i > indexCorte; i--) {
            bodyPCNet.removeChild(paragrafosPCNet[i]);
        }

        console.log('💉 Transplantando o Modelo Salvo...');
        const bodyModelo = docModelo.getElementsByTagName('w:body')[0];
        const pcnetSectPr = docPCNet.getElementsByTagName('w:sectPr')[0]; 

        for (let i = 0; i < bodyModelo.childNodes.length; i++) {
            const no = bodyModelo.childNodes[i];
            if (no.nodeName !== 'w:sectPr') {
                const noImportado = docPCNet.importNode(no, true);
                if (pcnetSectPr) {
                    bodyPCNet.insertBefore(noImportado, pcnetSectPr);
                } else {
                    bodyPCNet.appendChild(noImportado);
                }
            }
        }

        console.log('📦 Empacotando arquivo final...');
        const serializer = new XMLSerializer();
        const novoXml = serializer.serializeToString(docPCNet);
        zipPCNet.file("word/document.xml", novoXml);

        const bufferFinal = zipPCNet.generate({ type: 'nodebuffer', compression: 'DEFLATE' });

        res.setHeader('Content-Type', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document');
        res.setHeader('Content-Disposition', `attachment; filename="laudo_${tipoObjeto}_pronto.docx"`);
        return res.send(bufferFinal);

    } catch (error) {
        console.error('Erro ao gerar laudo:', error);
        res.status(500).json({ erro: error.message });
    }
});

app.listen(3000, () => {
    console.log('🚀 Servidor rodando na porta 3000');
});
